const NEXT_LABEL = {
  pending:'Equipment Assigned', process:'Strut Set', strutset:'Cutting Station',
  cutting:'Runner', runner:'Wood Shore Secured', secured:'Strut Equipment Returned',
};
const PREV_LABEL = {
  process:'Pending Equipment', strutset:'Equipment Assigned', cutting:'Strut Set',
  runner:'Cutting Station', secured:'Runner', returned:'Wood Shore Secured',
};
const VALUE_LABEL = {
  pending:'Required', process:'Required', strutset:'Required', waiting:'Required',
  cutting:'Cut length', runner:'Cut length', secured:'Set length', returned:'Set length',
};

/* Unicode vulgar fractions → stacked digit pairs; spaces → thin spaces. */
const FRAC_GLYPHS = {
  '\u00bc':[1,4], '\u00bd':[1,2], '\u00be':[3,4], '\u2153':[1,3], '\u2154':[2,3],
  '\u215b':[1,8], '\u215c':[3,8], '\u215d':[5,8], '\u215e':[7,8],
};
function MeasureText({ v }) {
  return <>{[...String(v)].map((ch, i) => {
    const f = FRAC_GLYPHS[ch];
    if (f) return <span className="fr" key={i}><span className="fr-n">{f[0]}</span><span className="fr-d">{f[1]}</span></span>;
    if (ch === ' ') return <span key={i}>{'\u2009'}</span>;
    return ch;
  })}</>;
}

function Arrow({ dir = 'right' }) {
  return (
    <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5"
      strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
      {dir === 'right' ? <path d="M9 6l6 6-6 6"/> : <path d="M15 6l-6 6 6 6"/>}
    </svg>
  );
}

/**
 * The gate primitive — a shore point with a status left-stripe, a
 * lifecycle-aware value (Required / Cut length / Set length), a
 * slide-to-advance control, and an always-available reverse (no
 * timed undo). Supports a Hazard badge and the removed/slashed state.
 */
export function ShorePointCard({
  title, location, apparatus, value, state = 'pending', hazard = false, active = false,
  removed = false, group, caption, onAdvance, onReverse,
}) {
  const trackRef = React.useRef(null);
  const [drag, setDrag] = React.useState(0);
  const next = NEXT_LABEL[state];
  const prev = PREV_LABEL[state];
  const promoted = state === 'cutting';
  const lineVar = { ['--_line']: `var(--st-${state}-line)` };

  const backRef = React.useRef(null);
  const [backDrag, setBackDrag] = React.useState(0);

  function startBackDrag(e) {
    if (!prev) return;
    e.preventDefault();
    const track = backRef.current;
    const max = track.clientWidth - 44;
    const startX = (e.touches ? e.touches[0].clientX : e.clientX);
    function move(ev) {
      const x = startX - (ev.touches ? ev.touches[0].clientX : ev.clientX);
      setBackDrag(Math.max(0, Math.min(max, x)));
    }
    function up(ev) {
      const x = startX - (ev.changedTouches ? ev.changedTouches[0].clientX : ev.clientX);
      window.removeEventListener('pointermove', move);
      window.removeEventListener('pointerup', up);
      if (x >= max * 0.6) { setBackDrag(max); setTimeout(() => { setBackDrag(0); onReverse && onReverse(); }, 120); }
      else setBackDrag(0);
    }
    window.addEventListener('pointermove', move);
    window.addEventListener('pointerup', up);
  }

  function startDrag(e) {
    if (!next) return;
    e.preventDefault();
    const track = trackRef.current;
    const max = track.clientWidth - 52;
    const startX = (e.touches ? e.touches[0].clientX : e.clientX);
    function move(ev) {
      const x = (ev.touches ? ev.touches[0].clientX : ev.clientX) - startX;
      setDrag(Math.max(0, Math.min(max, x)));
    }
    function up(ev) {
      const x = (ev.changedTouches ? ev.changedTouches[0].clientX : ev.clientX) - startX;
      window.removeEventListener('pointermove', move);
      window.removeEventListener('pointerup', up);
      if (x >= max * 0.6) { setDrag(max); setTimeout(() => { setDrag(0); onAdvance && onAdvance(); }, 120); }
      else setDrag(0);
    }
    window.addEventListener('pointermove', move);
    window.addEventListener('pointerup', up);
  }

  return (
    <div className={`fs-spc${active ? ' is-active' : ''}${removed ? ' is-removed' : ''}`} style={lineVar}>
      <div className="fs-spc__head">
        <div className="fs-spc__title">{title}</div>
        <div className="fs-spc__badges">
          {group && <span className="fs-spc__grpbadge">{group}</span>}
          <StatusInline state={state} />
          {hazard && <StatusInline state="danger" label="⚠ Hazard" />}
        </div>
      </div>
      {location && <div className="fs-spc__sub">{location}</div>}
      {apparatus && <div className="fs-spc__sub fs-spc__apparatus">{apparatus}</div>}
      {value && (
        <div className={`fs-spc__value${promoted ? ' is-promoted' : ''}`}>
          <span className="fs-spc__value-label">{VALUE_LABEL[state]}</span>
          <span className="fs-spc__value-num"><MeasureText v={value} /></span>
        </div>
      )}

      {state === 'waiting' && !removed && (
        <div className="fs-spc__wait">
          <span className="fs-spc__wait-ic">
            <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"
              strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
              <path d="M5 22h14M5 2h14M17 22v-4.17a2 2 0 0 0-.59-1.42L12 12l-4.41 4.41A2 2 0 0 0 7 17.83V22M7 2v4.17a2 2 0 0 0 .59 1.42L12 12l4.41-4.41A2 2 0 0 0 17 6.17V2"/>
            </svg>
          </span>
          <div>
            <div className="fs-spc__wait-t">Waiting for inventory</div>
            <div className="fs-spc__wait-d">No matching strut available to deploy — request equipment.</div>
          </div>
        </div>
      )}

      {!removed && next && (
        <div className="fs-spc__slide" ref={trackRef}>
          <span className="fs-spc__slide-label">Slide to set {next}</span>
          <div className="fs-spc__knob" style={{ transform: `translateX(${drag}px)` }}
            onPointerDown={startDrag} role="button" aria-label={`Slide to set ${next}`}>
            <Arrow dir="right" />
          </div>
        </div>
      )}
      {!removed && prev && (
        <div className="fs-spc__back" ref={backRef}>
          <span className="fs-spc__back-label">Slide back to {prev}</span>
          <div className="fs-spc__back-knob"
            style={{ transform: `translateY(-50%) translateX(-${backDrag}px)` }}
            onPointerDown={startBackDrag}
            role="button"
            aria-label={`Slide back to ${prev}`}>
            <Arrow dir="left" />
          </div>
        </div>
      )}

      {caption && <div className="fs-spc__caption">{caption}</div>}

      {removed && (<>
        <svg className="fs-spc__slash" preserveAspectRatio="none" viewBox="0 0 100 100">
          <line x1="100" y1="0" x2="0" y2="100" vectorEffect="non-scaling-stroke" />
        </svg>
        <span className="fs-spc__removed">Removed from cut list</span>
      </>)}
    </div>
  );
}

/* Inline status pill so the card is self-contained (mirrors StatusBadge). */
const STATUS_LABELS = { pending:'Pending Equipment', process:'Equipment Assigned', strutset:'Strut Set',
  cutting:'Cutting Station', runner:'Runner', secured:'Wood Shore Secured',
  returned:'Strut Equipment Returned', waiting:'Waiting', danger:'Hazard' };
function StatusInline({ state, label }) {
  return (
    <span style={{
      display:'inline-flex', alignItems:'center', gap:4, fontFamily:'var(--font-ui)',
      fontSize:'var(--type-caption)', fontWeight:600, lineHeight:1.4,
      padding:'4px 10px', borderRadius:'var(--radius-sm)', whiteSpace:'nowrap',
      color:`var(--st-${state}-ink)`, background:`var(--st-${state}-fill)`,
      border:`1px solid var(--st-${state}-line)`,
    }}>{label || STATUS_LABELS[state]}</span>
  );
}

export default ShorePointCard;
