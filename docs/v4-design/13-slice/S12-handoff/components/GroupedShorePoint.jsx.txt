// ShorePointCard is resolved from the compiled DS bundle at render time.
// The DS compiler does not support cross-component relative imports.
function resolveSPC() {
  const key = Object.keys(window).find(k => k.startsWith('FieldShoreDesignSystem_'));
  return key ? window[key].ShorePointCard : null;
}

function GsChevUp() {
  return (
    <svg viewBox="0 0 24 24" fill="none" stroke="currentColor"
      strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
      <path d="M18 15l-6-6-6 6" />
    </svg>
  );
}

/**
 * Collapsed rolodex stack ⇄ expanded member list for multi-card shore points
 * (Double T = 2 members, 3-Post = 3 members, etc.)
 *
 * Collapsed: front card shows active member; other members fall left as
 * status-striped labeled tabs. Tap a tab to rotate that member to the front.
 * Tap the card body to expand all members.
 *
 * Expanded: border-left indented list of all member ShorePointCards, each
 * fully interactive. "Stack" collapses back.
 */
export const GroupedShorePoint = function GroupedShorePoint({
  groupTitle,
  location,
  apparatus,
  members,
  defaultExpanded,
  onAdvance,
  onReverse,
}) {
  members = members || [];
  defaultExpanded = defaultExpanded || false;

  const SPC = resolveSPC();
  const [expanded, setExpanded] = React.useState(defaultExpanded);
  const [active, setActive] = React.useState(0);
  const [dir, setDir] = React.useState('r');
  const [rollKey, setRollKey] = React.useState(0);
  const n = members.length;

  function jumpTo(i) {
    if (i === active) return;
    const fwd = ((i - active + n) % n) <= n / 2;
    setDir(fwd ? 'r' : 'l');
    setActive(i);
    setRollKey(function(k) { return k + 1; });
  }

  function onCardTap(e) {
    if (e.target.closest('.fs-spc__slide, .fs-spc__back, .fs-spc__knob')) return;
    setExpanded(true);
  }

  function tabSlot(i) {
    var d = (i - active + n) % n;
    return (n - 1) - d;
  }

  function memberLoc(label) {
    if (!label) return location;
    return (label + (location ? ' \u00b7 ' + location : '')).replace(/ \u00b7 $/, '');
  }

  if (!SPC || n === 0) return null;

  var m = members[active] || {};

  if (!expanded) {
    return (
      <div className="fs-gs">
        <div
          className="fs-gs__wrap"
          onClick={onCardTap}
          role="button"
          aria-label={groupTitle + ' \u2014 ' + n + ' cards, showing ' + (m.label || '') + '. Tap to expand.'}
        >
          {members.map(function(p, i) {
            if (i === active) return null;
            var slot = tabSlot(i);
            return (
              <div
                key={p.id}
                className="fs-gs__tab"
                style={{ left: slot * TAB_W, zIndex: slot + 1, '--_pline': 'var(--st-' + (p.state || 'pending') + '-line)' }}
                role="button"
                aria-label={'Bring ' + p.label + ' to front'}
                onClick={function(e) { e.stopPropagation(); jumpTo(i); }}
              >
                <span>{p.label}</span>
              </div>
            );
          })}

          <div
            key={rollKey}
            style={{ marginLeft: (n - 1) * TAB_W }}
            className={'fs-gs__front' + (rollKey > 0 ? (dir === 'r' ? ' fs-gs__roll-r' : ' fs-gs__roll-l') : '')}
          >
            <SPC
              title={groupTitle}
              location={memberLoc(m.label)}
              apparatus={apparatus}
              value={m.value}
              state={m.state}
              group={(active + 1) + ' / ' + n}
              onAdvance={onAdvance ? function() { onAdvance(m.id); } : undefined}
              onReverse={onReverse ? function() { onReverse(m.id); } : undefined}
            />
          </div>
        </div>

        <div className="fs-gs__pager" style={{ marginLeft: (n - 1) * TAB_W }}>
          <div className="fs-gs__dots">
            {members.map(function(p, i) {
              var on = i === active;
              return (
                <span
                  key={p.id}
                  className={'fs-gs__dot' + (on ? ' is-on' : '')}
                  style={{
                    background: 'var(--st-' + (p.state || 'pending') + '-line)',
                    opacity: on ? 1 : 0.45,
                    width: on ? 22 : undefined,
                    borderRadius: on ? 4 : undefined,
                  }}
                />
              );
            })}
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className="fs-gs">
      <div className="fs-gs__open">
        <div className="fs-gs__openhead">
          <span className="fs-gs__opentitle">{groupTitle} \u00b7 {n} cards</span>
          <button className="fs-gs__colbtn" onClick={function() { setExpanded(false); }}>
            Stack <GsChevUp />
          </button>
        </div>
        {members.map(function(p, i) {
          return (
            <div key={p.id} className="fs-gs__member" style={{ '--i': i }}>
              <SPC
                title={groupTitle}
                location={memberLoc(p.label)}
                apparatus={apparatus}
                value={p.value}
                state={p.state}
                group={(i + 1) + ' / ' + n}
                onAdvance={onAdvance ? function() { onAdvance(p.id); } : undefined}
                onReverse={onReverse ? function() { onReverse(p.id); } : undefined}
              />
            </div>
          );
        })}
      </div>
    </div>
  );
};

export default GroupedShorePoint;
