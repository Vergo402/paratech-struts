const SYS_LABEL = { gold: 'Gold', grey: 'Grey', lockstroke: 'LockStroke' };

/* Unicode vulgar fractions → stacked digit pairs; spaces → thin spaces. */
const REC_FRAC = {
  '\u00bc':[1,4], '\u00bd':[1,2], '\u00be':[3,4], '\u2153':[1,3], '\u2154':[2,3],
  '\u215b':[1,8], '\u215c':[3,8], '\u215d':[5,8], '\u215e':[7,8],
};
function RecMeasure({ v }) {
  return <>{[...String(v)].map((ch, i) => {
    const f = REC_FRAC[ch];
    if (f) return <span className="fr" key={i}><span className="fr-n">{f[0]}</span><span className="fr-d">{f[1]}</span></span>;
    if (ch === ' ') return <span key={i}>{'\u2009'}</span>;
    return ch;
  })}</>;
}

/**
 * Strut recommendation with an inline deduction ledger. Stripe and
 * identity carry the strut-system color. The ledger leads
 * (Required − deductions = Effective); rated capacity is demoted to
 * a quiet secondary line, shown at the strut's extended length
 * (`capacityAt`) and rounded DOWN to the published rating at that
 * length. An optional extension block sits above the ledger.
 */
export function RecommendationCard({
  model, system = 'gold', systemLabel, product, spec,
  location, apparatus,
  fits = true, fitLabel,
  extension, extensionNote,
  required, deductions = [], effective, capacity, capacityAt,
  caption = null,
}) {
  const sysVar = { ['--_sys']: `var(--sys-${system})` };
  const ident = [
    systemLabel || SYS_LABEL[system], product, spec,
  ].filter(Boolean);

  return (
    <div className="fs-rec" style={sysVar}>
      <div className="fs-rec__head">
        <div>
          <div className="fs-rec__identity">
            <b>{systemLabel || SYS_LABEL[system]}</b>
            {model && <> · {model}</>}
          </div>
          {spec && <div className="fs-rec__connectors">{spec}</div>}
          {apparatus && <div className="fs-rec__apparatus">{apparatus}</div>}
          {location && <div className="fs-rec__loc">{location}</div>}
        </div>
        <span className={`fs-rec__fit${fits ? '' : ' fs-rec__fit--no'}`}>
          {fitLabel || (fits ? 'Fits' : 'Too short')}
        </span>
      </div>

      {extension && (
        <div className="fs-rec__ext">
          <span className="fs-rec__ext-tile">+</span>
          <div className="fs-rec__ext-amt"><RecMeasure v={extension} /><br /><span>extension</span></div>
          {extensionNote && <div className="fs-rec__ext-note">{extensionNote}</div>}
        </div>
      )}

      <div className="fs-rec__ledger">
        {required != null && (
          <div className="fs-rec__row"><span>Raw opening</span><span className="num"><RecMeasure v={required} /></span></div>
        )}
        {deductions.map((d, i) => (
          <div className="fs-rec__row fs-rec__row--ded" key={i}>
            <span>– {d.label}</span><span className="num"><RecMeasure v={d.value} /></span>
          </div>
        ))}
        {effective != null && (
          <div className="fs-rec__row fs-rec__row--eff"><span>Required strut length</span><span className="num"><RecMeasure v={effective} /></span></div>
        )}
      </div>

      {caption && <div className="fs-rec__caption">{caption}</div>}

      {capacity != null && (
        <div className="fs-rec__cap">
          <span className="fs-rec__cap-label">Rated capacity{capacityAt != null && <> at <RecMeasure v={capacityAt} /></>}</span>
          <span className="fs-rec__cap-val">{capacity}</span>
        </div>
      )}
    </div>
  );
}

export default RecommendationCard;
