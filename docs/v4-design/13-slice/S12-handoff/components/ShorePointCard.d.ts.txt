import * as React from 'react';

export type ShoreState =
  | 'pending' | 'process' | 'strutset' | 'cutting'
  | 'runner' | 'secured' | 'returned' | 'waiting';

/**
 * The gate primitive — a shore point in the incident lifecycle.
 * Status stripe, lifecycle-aware value label (Required / Cut length
 * / Set length), slide-to-advance control, and an always-available
 * reverse (no timed undo). Supports a hazard badge and the
 * "Removed from cut list" state.
 *
 * @startingPoint section="App Shell" subtitle="Shore point card with slide-to-advance" viewport="700x320"
 */
export interface ShorePointCardProps {
  /** e.g. "B-2 · Single Upright". */
  title: string;
  /** Division / side, e.g. "Division 2 · C side". */
  location?: string;
  /** Apparatus assigned to this point, e.g. "Eng 7". Renders on its own line below location. */
  apparatus?: string;
  /** Measurement value (the label adapts to state), e.g. `65½"`. */
  value?: string;
  /** Lifecycle state. "waiting" shows the inventory callout. @default "pending" */
  state?: ShoreState;
  /** Show the ⚠ Hazard badge. @default false */
  hazard?: boolean;
  /** Focus/selected styling (accent border). @default false */
  active?: boolean;
  /** Red-slash "Removed from cut list" state. @default false */
  removed?: boolean;
  /** Group counter badge, e.g. "1 / 3". Shown in the header alongside the status badge. */
  group?: string;
  /** Small explanatory caption under the controls. */
  caption?: string;
  /** Fired when the slide control commits to the next state. */
  onAdvance?: () => void;
  /** Fired when the reverse control is tapped. */
  onReverse?: () => void;
}

export function ShorePointCard(props: ShorePointCardProps): JSX.Element;
export default ShorePointCard;
